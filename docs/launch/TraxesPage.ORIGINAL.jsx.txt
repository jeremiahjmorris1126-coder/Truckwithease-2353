function useInView(ref) {
  const [seen, setSeen] = useState(false);
  useEffect(() => {
    if (!ref.current) return;
    const obs = new IntersectionObserver(([e]) => { if (e.isIntersecting) setSeen(true); }, { threshold: 0.12 });
    obs.observe(ref.current);
    return () => obs.disconnect();
  }, []);
  return seen;
}

function FadeIn({ children, delay = 0, up = true, style = {} }) {
  const ref = useRef(null);
  const seen = useInView(ref);
  return (
    <div ref={ref} style={{
      opacity: seen ? 1 : 0,
      transform: seen ? "translateY(0)" : up ? "translateY(32px)" : "translateY(0)",
      transition: `opacity 0.65s cubic-bezier(.22,1,.36,1) ${delay}ms, transform 0.65s cubic-bezier(.22,1,.36,1) ${delay}ms`,
      ...style,
    }}>
      {children}
    </div>
  );
}

// Animated counter
function Counter({ target, prefix = "", suffix = "", duration = 1800 }) {
  const [val, setVal] = useState(0);
  const ref = useRef(null);
  const started = useRef(false);
  const seen = useInView(ref);

  useEffect(() => {
    if (!seen || started.current) return;
    started.current = true;
    const isFloat = target % 1 !== 0;
    const steps = 60;
    let step = 0;
    const inc = setInterval(() => {
      step++;
      const progress = step / steps;
      const ease = 1 - Math.pow(1 - progress, 3);
      setVal(isFloat ? parseFloat((target * ease).toFixed(1)) : Math.round(target * ease));
      if (step >= steps) clearInterval(inc);
    }, duration / steps);
    return () => clearInterval(inc);
  }, [seen]);

  return <span ref={ref}>{prefix}{val.toLocaleString()}{suffix}</span>;
}
